% Wrap long lines in fenced code blocks instead of letting them run off
% the page -- pandoc's default Verbatim/Highlighting environments don't
% wrap, and this book quotes real command output and long log lines.
\usepackage{fvextra}
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{
  breaklines,breaksymbolleft={},
  fontsize=\small,
  commandchars=\\\{\}
}
\DefineVerbatimEnvironment{verbatim}{Verbatim}{
  breaklines,breaksymbolleft={},
  fontsize=\small
}

% Glossary - pure definitions without page numbers
% Using 'toc' to add to table of contents
% Using 'nonumberlist' to suppress page numbers (Index provides those)
% Using 'nopostdot' to remove trailing periods after descriptions
\usepackage[toc,nonumberlist,nopostdot]{glossaries}
\makeglossaries
\input{glossary.tex}
% Print all glossary entries even if not explicitly referenced
\glsaddall

% Index for page references
\usepackage{imakeidx}
\makeindex[intoc,columns=2,title=Index]
